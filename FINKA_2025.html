## Statistical Prep - DRY Version (Fixed)

# Define the groups you want to analyze
target_groups <- c("1", "2", "5", "6", "7", "8", "9")

# Define the variables you want statistics for
vars_to_summarise <- c("artenzahl_flora", "deckung_flora", "bluehende_pflanzen", 
                       "deckung_kultur", "hnv_taxa")

# Function to compute statistics
compute_stats <- function(data) { 
  data %>%
    group_by(group, jahr) %>%
    summarise(
      across(all_of(vars_to_summarise),
             list(
               mean = ~mean(.x, na.rm = TRUE),
               median = ~median(.x, na.rm = TRUE),
               sd = ~ifelse(length(na.omit(.x)) > 1, sd(.x, na.rm = TRUE), 0),
               min = ~min(.x, na.rm = TRUE),
               max = ~max(.x, na.rm = TRUE)
             ),
             .names = "{.col}_{.fn}"),
      .groups = "drop"
    ) %>%
    tidyr::complete(group, jahr)
}

# Function to create ordered factor levels for group_jahr
create_ordered_levels <- function(group_jahr_vector) {
  ord_df <- tibble(gj = unique(as.character(group_jahr_vector))) %>%
    mutate(
      prefix = str_extract(gj, "^[A-Za-z]+") %>% tolower(),
      year = str_extract(gj, "\\d{4}") %>% as.integer(),
      prefix_rank = case_when(
        prefix == "k" ~ 1L,
        prefix == "m" ~ 2L,
        prefix == "b" ~ 3L,
        TRUE ~ 4L
      )
    ) %>%
    arrange(is.na(year), year, prefix_rank, gj)
  
  return(ord_df$gj)
}

# Function to process each group
process_group <- function(data, group_number) {
  # Filter for specific kulturartengruppe
  filtered_data <- data %>% 
    filter(kulturartengruppe == group_number) %>%
    mutate(group_jahr = interaction(group, jahr, sep = "_", drop = TRUE))
  
  # Create ordered levels and apply them
  new_levels <- create_ordered_levels(filtered_data$group_jahr)
  filtered_data <- filtered_data %>%
    mutate(group_jahr = factor(as.character(group_jahr), levels = new_levels))
  
  return(filtered_data)
}

# Process all groups and create separate dataframes
gruppe_data <- target_groups %>%
  set_names(paste0("gruppe", .)) %>%
  map(~process_group(combined, .x))

# Compute statistics for all groups - FIXED VERSION
gruppe_stats_list <- target_groups %>%
  set_names(paste0("gruppe", ., "_stats")) %>%
  map(~{
    # Filter for this specific group
    data <- combined %>% filter(kulturartengruppe == .x)
    # Compute stats
    stats <- compute_stats(data)
    # Add group_jahr and apply ordered levels
    stats <- stats %>% mutate(group_jahr = interaction(group, jahr, sep = "_"))
    new_levels <- create_ordered_levels(stats$group_jahr)
    stats %>% mutate(group_jahr = factor(as.character(group_jahr), levels = new_levels))
  })

    summarise(total_count = sum(count), .groups = "drop")
  
  # Order x-axis (group_jahr)
  ord_levels <- hnv_counts %>%
    distinct(group_jahr, jahr, group) %>%
    arrange(jahr, factor(group, levels = c("k", "m", "b"))) %>%
    pull(group_jahr)
  
  hnv_counts <- hnv_counts %>%
    mutate(group_jahr = factor(group_jahr, levels = ord_levels))
  
  hnv_totals <- hnv_totals %>%
    mutate(group_jahr = factor(group_jahr, levels = ord_levels))
  
  list(counts = hnv_counts, totals = hnv_totals)
}

# Process all groups at once
hnv_results <- target_groups %>%
  set_names(paste0("gruppe", .)) %>%
  map(~process_hnv_data(combined_mutate, .x))

# Extract counts and totals for each group
hnv_counts_list <- hnv_results %>%
  map("counts") %>%
  set_names(paste0("hnv_counts_g", target_groups))

hnv_totals_list <- hnv_results %>%
